cmake_minimum_required(VERSION 3.22.1)
project(PhotoStudioRealESRGAN)

file(GLOB_RECURSE NCNN_CONFIG_FILES "${CMAKE_CURRENT_SOURCE_DIR}/*/ncnnConfig.cmake")
set(ncnn_DIR "")
foreach(NCNN_CONFIG_FILE ${NCNN_CONFIG_FILES})
    if(NCNN_CONFIG_FILE MATCHES "/${ANDROID_ABI}/lib/cmake/ncnn/ncnnConfig.cmake")
        get_filename_component(ncnn_DIR "${NCNN_CONFIG_FILE}" DIRECTORY)
        break()
    endif()
endforeach()
if(NOT ncnn_DIR)
    message(FATAL_ERROR "NCNN Vulkan CMake package not found for ABI ${ANDROID_ABI}. Inspect the downloaded ncnn release archive layout.")
endif()
find_package(ncnn REQUIRED)

add_library(photostudio_realesrgan SHARED realesrgan_jni.cpp)
target_compile_features(photostudio_realesrgan PRIVATE cxx_std_17)
target_compile_options(photostudio_realesrgan PRIVATE -O3)
target_link_libraries(photostudio_realesrgan PRIVATE ncnn android log)
